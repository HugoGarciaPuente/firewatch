# Dependencies shared by every tier.
#
# This is the ONLY file all three targets install. It exists to make the
# shared base explicit rather than duplicated three times by hand, where a
# bump would eventually be applied to two targets and not the third.
#
# Read requirements/README.md before editing any of these. The split is
# enforced by test_api.py::test_tier_requirements_cover_imports, which fails if
# a tier's image would be missing something its own code imports.

# Settings loading (os.environ / .env parsing). config.py imports it in all
# three tiers, so it is genuinely universal.
python-dotenv==1.2.4

# NASA FIRMS + Open-Meteo HTTP client. Needed transitively by all three tiers:
#   acquisition -> clients/firms.py, clients/weather.py
#   processing  -> services/processing.py imports services/risk.py, which
#                  enriches records through clients/weather.py
#   api         -> clients/firms.py, imported by the /fire/ingest endpoint
# Note this means the api image carries an outbound HTTP client it does not
# call itself. That is a real coupling, recorded rather than hidden.
httpx==0.28.1

# Kafka producer/consumer.
#   acquisition -> services/ingestion.py (produces)
#   processing  -> services/processing.py (consumes)
#   api         -> services/ingestion.py, reached via the /fire/ingest endpoint
kafka-python==3.0.11
